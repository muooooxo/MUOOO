function genOrderId(){
  const d = new Date();
  return "M" + d.getFullYear().toString().slice(-2) + 
         String(d.getMonth()+1).padStart(2,"0") + 
         String(d.getDate()).padStart(2,"0") + "-" + 
         String(d.getHours()).padStart(2,"0") + 
         String(d.getMinutes()).padStart(2,"0") + "-" +

}

function calcAll() {
  let gSum = 0; 
  let normalQtyTotal = 0; 
  let normalCountForServiceFee = 0; 
  const RATE_UNIFIED = 5.5; 

  // 1. 計算商品金額
  document.querySelectorAll('.product-box').forEach(p => {
    const price = parseFloat(p.querySelector('.p-price').value) || 0;
    const qty = parseInt(p.querySelector('.p-qty').value) || 1;
    const shipping = parseFloat(p.querySelector('.p-shipping').value) || 0; 
    const isAddon = p.dataset.type === "addon";
    let sub = 0;
    
    if (price > 0) {
      if (price >= 400) { 
        sub = (price * qty + shipping) * RATE_UNIFIED; 
        if (!isAddon) normalQtyTotal += qty;
      } 
      else {
        if (isAddon) { 
          sub = ((price * qty) + shipping) * RATE_UNIFIED + (30 * qty); 
        } 
        else { 
          normalQtyTotal += qty;
          for(let i=0; i<qty; i++) {
            normalCountForServiceFee++;
            sub += (price * RATE_UNIFIED) + (normalCountForServiceFee === 1 ? 80 : 70);
          }
          sub += (shipping * RATE_UNIFIED);
        }
      }
    }
    const roundedSub = Math.ceil(sub);
    p.querySelector('.item-total').textContent = `報價：NT$ ${roundedSub.toLocaleString()}`;
    gSum += roundedSub;
  });

  // 2. 多件滿件折抵優惠
  let discount = 0;
  let discountMsg = "";
  
  if (normalQtyTotal >= 10) {
    discount = 120;
    discountMsg = "🎉 自助滿10件折 $120";
  } else if (normalQtyTotal >= 5) {
    discount = 50;
    discountMsg = "🎉 自助滿5件折 $50";
  } else if (normalQtyTotal >= 3) {
    discount = 20;
    discountMsg = "🎉 自助滿3件折 $20";
  }

  // 3. 計算附加服務金額
  let extraServiceTotal = 0;
  document.querySelectorAll('.extra-service-cb:checked').forEach(cb => {
    if (cb.dataset.type === "per_item") {
      const itemQty = parseInt($('addonCountQty').value) || 1;
      extraServiceTotal += (parseInt(cb.dataset.price) || 0) * itemQty;
    } else {
      extraServiceTotal += parseInt(cb.dataset.price) || 0;
    }
  });

  // UI 顯示控制
  if (discount > 0) {
    $('discountArea').style.display = 'block';
    $('discountTag').textContent = discountMsg;
    $('originalTotal').style.display = 'block';
    $('originalTotal').textContent = "NT$ " + (gSum + extraServiceTotal).toLocaleString();
  } else {
    $('discountArea').style.display = 'none';
    $('originalTotal').style.display = 'none';
  }

  // 最終第一階段金額
  const finalGoods = gSum - discount + extraServiceTotal;
  $('goodsTotal').textContent = "NT$ " + (finalGoods > 0 ? finalGoods : 0).toLocaleString();
}

function updateShippingUI() {
  const mode = $('intlMode').value;
  const isDirect = mode.includes('direct');
  
  $('shippingArea').style.display = isDirect ? 'block' : 'none';
  $('ezwayNote').style.display = isDirect ? 'block' : 'none';
  
  if (isDirect) {
    const isCvs = $('twMode').value === 'cvs';
    $('cvsNameField').style.display = isCvs ? 'block' : 'none';
    if (isCvs) { $('cvsName').classList.add('required'); } 
    else { $('cvsName').classList.remove('required'); $('cvsName').classList.remove('error-border'); }
  }
}

function submitOrder() {
  let hasError = false;
  document.querySelectorAll('.required').forEach(el => {
    if (el.offsetParent !== null && !el.value.trim()) {
      el.classList.add('error-border');
      hasError = true;
    } else {
      el.classList.remove('error-border');
    }
  });

  if (hasError) {
    alert("❌ 還有必填欄位沒寫到喔！");
    const errEl = document.querySelector('.error-border');
    if(errEl) errEl.scrollIntoView({ behavior: 'smooth', block: 'center' });
    return;
  }

  let pDetails = "";
  document.querySelectorAll('.product-box').forEach((p, index) => {
    const rmb = p.querySelector('.p-price').value;
    const qty = p.querySelector('.p-qty').value;
    const shipping = p.querySelector('.p-shipping').value || '0';
    const sub = p.querySelector('.item-total').textContent;
    const spec = p.querySelector('.p-spec').value || '無';
    const memo = p.querySelector('.p-memo').value || '無';
    const typeTag = p.dataset.type === "addon" ? "【加購商品】" : "【一般商品】";
    
    const pName = p.querySelector('.p-name').value.trim() || `未命名商品 #${index + 1}`;
    
    pDetails += `${typeTag} ${pName}\n    規格:${spec} | 備註:${memo}\n    明細:RMB ${rmb} x ${qty} (郵費:${shipping}) | ${sub}\n    網址:${p.querySelector('.p-link').value}\n\n`;
  });

  // 整理勾選的附加服務文字
  let selectedServices = [];
  document.querySelectorAll('.extra-service-cb:checked').forEach(cb => {
    if (cb.dataset.type === "per_item") {
      const itemQty = parseInt($('addonCountQty').value) || 1;
      const totalCost = itemQty * 2;
      selectedServices.push(`${cb.dataset.name} x${itemQty}件 (+NT$ ${totalCost})`);
    } else {
      const priceVal = parseInt(cb.dataset.price);
      const priceText = priceVal > 0 ? ` (+NT$ ${priceVal})` : " (二補計費)";
      selectedServices.push(`${cb.dataset.name}${priceText}`);
    }
  });
  const extraServicesText = selectedServices.length > 0 ? selectedServices.join("\n    - ") : "無加購";

  const modeSel = $('intlMode');
  const modeText = modeSel.options[modeSel.selectedIndex].text;
  
  const isDirect = modeSel.value.includes('direct');
  const shippingInfo = isDirect 
    ? `
👤 直寄收件資料：
${$('realName').value}
${$('mobile').value}
${$('idNo').value}
${$('addr').value}
${$('twMode').value === 'cvs' ? '門市：'+$('cvsName').value : '方式：宅配'}`
    : `👤 收件方式：轉寄 `;

  const activeDiscount = $('discountArea').style.display === 'block' ? `(${$('discountTag').textContent})` : "";

  const summary = `【 MUOOO 自助訂單摘要 】
單號：${orderId}
會員：${$('memberName').value} / ${$('memberPhone').value}
-----------------------
💰 應付金額 ：${$('goodsTotal').textContent} ${activeDiscount}
🛠️ 附加服務 ：
    - ${extraServicesText}
(⚠️ 付款回報：請至 LINE 輸入關鍵字『下單付款』)

🚚 運輸選擇：${modeText}
${shippingInfo}
-----------------------
📦 商品明細 (含原價/報價/備註)：
${pDetails}
-----------------------
💡 訂單已建立，請輸入關鍵字「下單付款」進行付款與對帳。`;

  // 1. 同步背景資料庫 (Google Forms)
  const memberIdentity = `會員:${$('memberName').value} / 電話:${$('memberPhone').value}`;
  const body = new URLSearchParams({
    "entry.1746867203": orderId,
    "entry.1748508385": summary,
    "entry.751786061": $('goodsTotal').textContent,
    "entry.1327908673": "0", 
    "entry.595640404": "0", 
    "entry.1656931618": memberIdentity 
  });

  if (isLiffInitialized && liff.isInClient()) {

      {
        type: 'text',
        text: summary

    ]).then(() => {
      alert("🎉 訂單摘要已自動傳送到 LINE 聊天室！請於聊天室輸入「下單付款」。");
      liff.closeWindow(); // 發送完畢自動關閉頁面

      console.error("LIFF 發送失敗", err);
